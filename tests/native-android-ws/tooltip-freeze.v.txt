// Recurse through hidden directories: frozen external inputs include .local.
fn all_files(root string) ![]string {
	mut result := []string{}
	mut names := os.ls(root)!
	names.sort()
	for name in names {
		path := root.trim_right('/') + '/' + name
		assert !os.is_link(path), path
		if os.is_dir(path) { result << all_files(path)! }
		else { assert os.is_file(path), path; result << path }
	}
	return result
}

fn inventory_self_test() ! {
	root := os.temp_dir() + '/tooltip-inventory-' + os.getpid().str() + '-' + time.utc().unix_micro().str()
	assert !os.exists(root)
	os.mkdir_all(root + '/.hidden/nested')!
	defer { os.rmdir_all(root) or {} }
	os.write_file(root + '/plain', 'plain')!
	os.write_file(root + '/.hidden/nested/input', 'hidden')!
	mut actual := all_files(root)!.map(it.all_after(root + '/'))
	actual.sort()
	assert actual == ['.hidden/nested/input', 'plain']
	assert all_files(root + '/')! == all_files(root)!
	println('TOOLTIP' + attempt + ' HIDDEN INVENTORY SELF-TEST PASS')
}

fn check_sources(kind string) ! {
	directory := if kind == 'framework' { '/sources/' } else { '/atlframework/' }
	mut actual := map[string]string{}
	for path in all_files(base + '/' + kind + directory)!.filter(it.ends_with('.v')) {
		actual[path.all_after(base + '/' + kind + directory)] = hash(path)!
	}
	assert actual == json2.decode[map[string]string](os.read_file(base + '/' + kind + '/sources.json')!)!, kind
}
