	}

fn host_getsockopt(fd i32, level i32, option i32, value voidptr, length &u32) i32 {

fn host_send(fd i32, bytes voidptr, length usize, flags i32) isize {
	assert fd == 23 && flags == 0x4000
	// A failed synchronous connect never creates a stream, even if SO_ERROR is zero.
	if transfer_failed || (connect_error != 0 && connect_error != C.EINPROGRESS)
		|| (connect_error == C.EINPROGRESS && async_error != 0) { return -1 }
	unsafe { for i in 0 .. int(length) { written << (&u8(bytes))[i] }
	 }
	return isize(length)
}

fn host_recv(fd i32, bytes voidptr, length usize, flags i32) isize {
	assert fd == 23 && flags == 0
	if transfer_failed || received == reply.len { return 0 }
	// Exercise partial reads through the actual production transfer loop.
	count := if length > 1 { 1 } else { int(length) }
	unsafe { for i in 0 .. count { (&u8(bytes))[i] = reply[received + i] }
	 }
	received += count
	return count
}

			assert poll_calls == 1 && getsockopt_calls == 1 && written.len == 0
		}

			reset()
