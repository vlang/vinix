struct Started { pid int session u32 }

fn discard_manager_error(err IError) {
	message := err.msg()
	unsafe { message.free(); err.free() }
}
fn report_manager_error(err IError, prefix string) { discard_manager_error(err) }

fn wait(pid int) int {
	mut status := i32(0)
	for C.waitpid(pid, &status, 0) < 0 {
		if C.errno != C.EINTR { return 1 }
	}
	return int(status)
}

fn start_fallback(db accountcore.Database, user accountcore.User, argv []string, fallback []string, seat bool,
	control int, unlock Unlock) !Started {
	if fixture_mode == 3 { return error('controlled launch refusal') }
	pid := C.fork()
	assert pid >= 0
	if pid == 0 {
		mut byte := u8(0)
		assert C.read(control, &byte, 1) == 1 && byte == `C`
		// The parent's wipe happens after this fork. Its child must retain
		// the original bytes until the keyring handoff, without a shared alias.
		assert unlock.secret.len == 16 && unlock.secret.all(it == `s`)
		byte = `Y`
		assert C.write(control, &byte, 1) == 1
		// Force a real asynchronous retirement window. The old WNOHANG
		// cleanup returns before this child exits and leaves it unreaped.
		C.usleep(150_000)
		C._exit(0)
	}
	spawned = pid
	return Started{pid: pid, session: u32(pid)}
}

fn readable(fd int, milliseconds int) bool {
	assert fixture_password.len == 16 && fixture_password.all(it == 0)
	mut byte := u8(`C`)
	assert C.write(fd, &byte, 1) == 1
	assert C.read(fd, &byte, 1) == 1 && byte == `Y`
	attendance_observed = true
	manager_signal = 1
	return false
}

fn main() {
	for scenario in 0 .. 5 {
		fixture_mode = scenario
		manager_signal = 0
		attendance_observed = false
		spawned = 0
		fixture_password = []u8{len: 16, init: `s`}
		mut manager := Manager{config: ManagerConfig{}}
		manager.sessions.flags |= .noslices
		user := accountcore.User{name: 'alice', uid: 1000, gid: 1000,
			home: '/home/alice', shell: '/bin/sh'}
		if scenario == 1 || scenario == 4 {
			mut pair := [2]i32{}
			assert C.socketpair(C.AF_UNIX, C.SOCK_STREAM, 0, &pair[0]) == 0
			started := start_fallback(accountcore.Database{}, user, [], [], true, int(pair[1]), Unlock{secret: fixture_password}) or { panic(err) }
			C.close(pair[1])
			manager.sessions << Session{user: user.owned_copy(), pid: started.pid,
				session: started.session, control: int(pair[0]), stopped: true}
		}
		manager.attend(user, fixture_password)
		assert fixture_password.all(it == 0)
		assert attendance_observed == (scenario <= 1)
		if scenario <= 1 {
			mut status := i32(0)
			assert C.waitpid(spawned, &status, C.WNOHANG) == -1 && C.errno == C.ECHILD
			assert manager.sessions.len == 0
		}
		if scenario == 4 {
			C.kill(spawned, C.SIGKILL)
			C.close(manager.sessions[0].control)
			manager.sessions[0].user.close()
		}
		if spawned != 0 {
			mut status := i32(0)
			got := C.waitpid(spawned, &status, 0)
			if got == spawned && scenario != 4 { assert status == 0 }
		}
		unsafe { fixture_password.free(); manager.sessions.free() }
	}
	println('PASS manager password wiped before attendance: new, resumed, database, launch and seat refusal; forked child retains handoff copy; direct child reaped before next admission')
}
