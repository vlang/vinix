fn root_path(view i32) &char {
	unsafe { return if view == 0 { &char(c'/proc') } else if view == 1 { &char(c'/tmp/linux-procself/alias') } else { &char(c'/tmp/linux-procself/jail/view') } }
}
fn link_name(kind i32) &char { unsafe { return if kind == 0 { &char(c'self') } else { &char(c'thread-self') } } }
fn cleanup(bank &Handles) {
	unsafe { for fd in bank.links { if fd >= 0 { C.close(fd) } }; for fd in bank.dirs { if fd >= 0 { C.close(fd) } } }
}
fn prepare(bank &Handles) bool {
	unsafe {
		mut handles := &Handles(bank)
		for view in 0 .. 3 {
			handles.dirs[view] = C.open(root_path(i32(view)), C.O_PATH | C.O_DIRECTORY)
			if handles.dirs[view] < 0 { return false }
			for kind in 0 .. 2 {
				handles.links[view * 2 + kind] = C.openat(handles.dirs[view], link_name(i32(kind)), C.O_PATH | C.O_NOFOLLOW)
				if handles.links[view * 2 + kind] < 0 { return false }
			}
		}
		return true
	}
}
fn probe(bank &Handles, view i32, kind i32, form i32, size i32, destination i32, verbose bool) bool {
	unsafe {
		pid := C.getpid(); tid := i32(C.syscall(isize(C.SYS_gettid)))
		mut target := [64]char{}; mut path := [128]char{}
		length := if kind == 0 { C.snprintf(&target[0], 64, c'%d', pid) } else { C.snprintf(&target[0], 64, c'%d/task/%d', pid, tid) }
		if length <= 0 || length >= 64 || C.snprintf(&path[0], 128, c'%s/%s', root_path(view), link_name(kind)) >= 128 { return false }
		mut bytes := [64]u8{}; for index in 0 .. 64 { bytes[index] = 0x5a }
		buf := if destination == 0 { voidptr(&bytes[0]) } else if destination == 1 { voidptr(nil) } else { voidptr(usize(1)) }
		e := C.__errno_location(); *e = 0
		actual := if form == 0 { C.syscall(isize(C.SYS_readlinkat), bank.dirs[view], link_name(kind), buf, size) }
			else if form == 1 { C.readlinkat(bank.dirs[view], link_name(kind), buf, usize(size)) }
			else if form == 2 { C.syscall(isize(C.SYS_readlinkat), bank.links[view * 2 + kind], c'', buf, size) }
			else if form == 3 { C.readlinkat(bank.links[view * 2 + kind], c'', buf, usize(size)) }
			else { C.readlink(&path[0], buf, usize(size)) }; code := *e
		copied := if size < length { size } else { length }
		wanted := if destination == 0 { isize(copied) } else { isize(-1) }; error_code := if destination == 0 { 0 } else { C.EFAULT }
		mut valid := actual == wanted && code == error_code
		for index in 0 .. 64 { if bytes[index] != if destination == 0 && index < copied { u8(target[index]) } else { u8(0x5a) } { valid = false } }
		if !valid && verbose { C.printf(c'LINUX PROCSELF DETAIL: pid=%d tid=%d view=%d kind=%d form=%d size=%d dest=%d result=%ld/%d expected=%ld/%d\n', pid, tid, view, kind, form, size, destination, actual, code, wanted, error_code); C.fflush(nil) }
		return valid
	}
}
fn scheduling(policy i32, priority i32) bool {
	unsafe {
		param := C.sched_param{sched_priority: priority}; mut actual := C.sched_param{}; mut actual_policy := i32(-1); current := C.pthread_self()
		return C.pthread_setschedparam(current, policy, &param) == 0 && C.pthread_getschedparam(current, &actual_policy, &actual) == 0 && actual_policy == policy && actual.sched_priority == priority
	}
}
fn usage_ticks(value C.timeval) i64 { return value.tv_sec * 100 + value.tv_usec / 10000 }
fn workload(slot i32) bool {
	unsafe {
		page := C.sysconf(C._SC_PAGESIZE); if page <= 0 { return false }
		pages := if slot == 0 { i32(256) } else { i32(slot * 32) }
		// Large reservations demand-fill on both architectures. ARM eagerly
		// fills small anonymous mappings, so touching those proves no faults.
		length := usize(64 * 1024 * 1024)
		mut fault_before := C.rusage{}; mut fault_after := C.rusage{}
		if C.getrusage(C.RUSAGE_THREAD, &fault_before) != 0 { return false }
		memory := C.mmap(nil, length, C.PROT_READ | C.PROT_WRITE, C.MAP_PRIVATE | C.MAP_ANONYMOUS, -1, 0)
		if memory == voidptr(isize(-1)) { return false }
		for index in 0 .. int(pages) { (&u8(memory))[index * int(page)] = u8(index) }
		if C.munmap(memory, length) != 0 { return false }
		if C.getrusage(C.RUSAGE_THREAD, &fault_after) != 0 { return false }
		faults := fault_after.ru_minflt - fault_before.ru_minflt
		if faults < pages || fault_after.ru_majflt != fault_before.ru_majflt { return false }
		mut first := C.timespec{}; mut now := C.timespec{}
		if C.clock_gettime(C.CLOCK_THREAD_CPUTIME_ID, &first) != 0 { return false }
		span := if slot == 0 { i64(200000000) } else { i64(slot * 50000000) }
		mut checksum := u64(1)
		for {
			for index in 0 .. 100000 { checksum = (checksum * 6364136223846793005 + 1) ^ u64(index) }
			if C.clock_gettime(C.CLOCK_THREAD_CPUTIME_ID, &now) != 0 { return false }
			if (now.tv_sec - first.tv_sec) * 1000000000 + now.tv_nsec - first.tv_nsec >= span { break }
		}
		C.printf(c'LINUX PROCSELF WORK: slot=%d pages=%d faults=%ld cpu_ns=%ld checksum=%lu\n', slot, pages, faults, (now.tv_sec - first.tv_sec) * 1000000000 + now.tv_nsec - first.tv_nsec, checksum); C.fflush(nil)
		return true
	}
}
fn stat_record(dirfd i32, path &char, wanted_pid i32, policy i32, rtprio i32, task bool, verbose bool) bool {
	unsafe {
		fd := C.openat(dirfd, path, C.O_RDONLY); if fd < 0 { return false }; defer { C.close(fd) }
		mut before := C.rusage{}; mut after := C.rusage{}; who := if task { i32(C.RUSAGE_THREAD) } else { i32(C.RUSAGE_SELF) }
		if C.getrusage(who, &before) != 0 { return false }
		mut data := [4096]char{}; length := C.read(fd, &data[0], 4095)
		if C.getrusage(who, &after) != 0 { return false }
		if length <= 0 || length >= 4096 { return false }; data[int(length)] = 0
		mut pid := i32(-1); if C.sscanf(&data[0], c'%d', &pid) != 1 { return false }
		end_name := C.strrchr(&data[0], i32(`)`)); if end_name == nil || end_name[1] != char(32) || end_name[2] == 0 || end_name[3] != char(32) { return false }
		mut cursor := end_name + 4; mut priority := i64(0); mut actual_rtprio := i64(0); mut actual_policy := i64(-1)
		mut minor := i64(-1); mut major := i64(-1); mut user := i64(-1); mut system := i64(-1)
		for field in 4 .. 42 {
			mut next := &char(nil); value := C.strtoll(cursor, &next, 10); if next == cursor { return false }; cursor = next
			if field == 18 { priority = value }; if field == 40 { actual_rtprio = value }; if field == 41 { actual_policy = value }
			if field == 10 { minor = value }; if field == 12 { major = value }; if field == 14 { user = value }; if field == 15 { system = value }
		}
		wanted_priority := if policy == i32(C.SCHED_FIFO) || policy == i32(C.SCHED_RR) { i64(-1 - rtprio) } else { i64(20) }
		identity := pid == wanted_pid && priority == wanted_priority && actual_rtprio == rtprio && actual_policy == policy
		accounting := minor >= before.ru_minflt && minor <= after.ru_minflt && major >= before.ru_majflt && major <= after.ru_majflt
			&& user >= usage_ticks(before.ru_utime) && user <= usage_ticks(after.ru_utime) && system >= usage_ticks(before.ru_stime) && system <= usage_ticks(after.ru_stime)
		valid := identity && accounting
		if !identity && verbose { C.printf(c'LINUX PROCSELF STAT DETAIL: path=%s pid=%d/%d priority=%ld/%ld rtprio=%ld/%d policy=%ld/%d\n', path, pid, wanted_pid, priority, wanted_priority, actual_rtprio, rtprio, actual_policy, policy); C.fflush(nil) }
		if !accounting && verbose { C.printf(c'LINUX PROCSELF ACCOUNT DETAIL: path=%s task=%d minor=%ld/%ld:%ld major=%ld/%ld:%ld user=%ld/%ld:%ld system=%ld/%ld:%ld\n', path, i32(task), minor, before.ru_minflt, after.ru_minflt, major, before.ru_majflt, after.ru_majflt, user, usage_ticks(before.ru_utime), usage_ticks(after.ru_utime), system, usage_ticks(before.ru_stime), usage_ticks(after.ru_stime)); C.fflush(nil) }
		return valid
	}
}
fn status(dirfd i32, kind i32, verbose bool) bool {
	unsafe {
		pid := C.getpid(); tid := if kind == 0 { pid } else { i32(C.syscall(isize(C.SYS_gettid))) }
		mut named := [64]char{}; mut numeric := [64]char{}
		C.snprintf(&named[0], 64, c'%s/status', link_name(kind))
		if kind == 0 { C.snprintf(&numeric[0], 64, c'%d/status', pid) } else { C.snprintf(&numeric[0], 64, c'%d/task/%d/status', pid, tid) }
		fd := C.openat(dirfd, &named[0], C.O_RDONLY); if fd < 0 { return false }; defer { C.close(fd) }
		explicit_fd := C.openat(dirfd, &numeric[0], C.O_RDONLY); if explicit_fd < 0 { return false }; defer { C.close(explicit_fd) }
		mut left := C.stat{}; mut right := C.stat{}; mut data := [4096]char{}
		if C.fstat(fd, &left) != 0 || C.fstat(explicit_fd, &right) != 0 || left.st_ino != right.st_ino || left.st_dev != right.st_dev || left.st_mode != right.st_mode { return false }
		length := C.read(fd, &data[0], 4095); if length <= 0 || length >= 4096 { return false }; data[int(length)] = 0
		tgid_line := C.strstr(&data[0], c'\nTgid:'); pid_line := C.strstr(&data[0], c'\nPid:'); ns_line := C.strstr(&data[0], c'\nNSpid:')
		mut actual_tgid := i32(-1); mut actual_pid := i32(-1); mut actual_ns := i32(-1); mut extra := i32(-1)
		valid_status := tgid_line != nil && pid_line != nil && ns_line != nil
			&& C.sscanf(tgid_line, c'\nTgid: %d', &actual_tgid) == 1 && actual_tgid == pid
			&& C.sscanf(pid_line, c'\nPid: %d', &actual_pid) == 1 && actual_pid == tid
			&& C.sscanf(ns_line, c'\nNSpid: %d %d', &actual_ns, &extra) == 1 && actual_ns == tid
		policy := i32(C.syscall(isize(C.SYS_sched_getscheduler), isize(tid))); mut param := C.sched_param{}
		if policy < 0 || C.syscall(isize(C.SYS_sched_getparam), isize(tid), &param) != 0 { return false }
		C.snprintf(&named[0], 64, c'%s/stat', link_name(kind))
		if kind == 0 { C.snprintf(&numeric[0], 64, c'%d/stat', pid) } else { C.snprintf(&numeric[0], 64, c'%d/task/%d/stat', pid, tid) }
		first := stat_record(dirfd, &named[0], tid, policy, param.sched_priority, kind == 1, verbose)
		second := stat_record(dirfd, &numeric[0], tid, policy, param.sched_priority, kind == 1, verbose)
		return valid_status && first && second
	}
}
fn matrix(bank &Handles, ordinary bool, slot i32) bool {
	unsafe {
		mut own := C.rusage{}; mut group := C.rusage{}
		if C.getrusage(C.RUSAGE_THREAD, &own) != 0 || C.getrusage(C.RUSAGE_SELF, &group) != 0 { return false }
		own_ticks := usage_ticks(own.ru_utime) + usage_ticks(own.ru_stime); group_ticks := usage_ticks(group.ru_utime) + usage_ticks(group.ru_stime)
		if slot != 0 && (group.ru_minflt <= own.ru_minflt || group_ticks - own_ticks < 10) { return false }
		C.printf(c'LINUX PROCSELF ACCOUNT SAMPLE: ordinary=%d slot=%d faults=%ld/%ld cpu_ticks=%ld/%ld\n', i32(ordinary), slot, own.ru_minflt, group.ru_minflt, own_ticks, group_ticks); C.fflush(nil)
		mut cases := i32(0); mut bad := i32(0)
		for view in 0 .. 3 { for kind in 0 .. 2 { for form in 0 .. 5 { for size in [i32(1), 3, 64]! { for destination in 0 .. 3 {
			cases++; if !probe(bank, i32(view), i32(kind), i32(form), size, i32(destination), true) { bad++ }
		} } } } }
		mut controls := true
		for view in 0 .. 3 { for kind in 0 .. 2 { if !status(bank.dirs[view], i32(kind), true) { controls = false } } }
		C.printf(c'LINUX PROCSELF OUTCOMES: ordinary=%d slot=%d pid=%d tid=%ld cases=%d failures=%d status=%d\n', i32(ordinary), slot, C.getpid(), C.syscall(isize(C.SYS_gettid)), cases, bad, i32(controls)); C.fflush(nil)
		return bad == 0 && controls
	}
}
fn worker(arg voidptr) voidptr {
	unsafe {
		mut state := &Worker(arg); state.tid = i32(C.syscall(isize(C.SYS_gettid)))
		if !workload(state.slot) { state.failed = true; return nil }
		policy := if state.slot == 1 { i32(C.SCHED_FIFO) } else { i32(C.SCHED_RR) }; priority := if state.slot == 1 { i32(17) } else { i32(33) }
		if !scheduling(policy, priority) { state.failed = true; return nil }
		state.failed = !matrix(state.bank, state.ordinary, state.slot)
		if !scheduling(i32(C.SCHED_OTHER), 0) { state.failed = true }; return nil
	}
}
fn state_name(code u8) &char {
	unsafe { return if code == `R` { &char(c'R (running)\n') } else if code == `S` { &char(c'S (sleeping)\n') } else if code == `T` { &char(c'T (stopped)\n') } else { &char(c'Z (zombie)\n') } }
}
fn state_record(dirfd i32, path &char, status_file bool, wanted u8, verbose bool) bool {
	unsafe {
		fd := C.openat(dirfd, path, C.O_RDONLY); if fd < 0 { return false }; defer { C.close(fd) }
		mut data := [4096]char{}; length := C.read(fd, &data[0], 4095)
		if length <= 0 || length >= 4096 { return false }; data[int(length)] = 0
		mut actual := u8(0); mut description := true
		if status_file {
			line := C.strstr(&data[0], c'\nState:\t'); if line == nil { return false }
			actual = u8(line[8]); description = C.strncmp(line + 8, state_name(wanted), C.strlen(state_name(wanted))) == 0
		} else {
			end_name := C.strrchr(&data[0], i32(`)`)); if end_name == nil || usize(end_name) + 4 > usize(&data[0]) + usize(length) || end_name[1] != char(32) || end_name[3] != char(32) { return false }
			actual = u8(end_name[2])
		}
		valid := actual == wanted && description
		if !valid && verbose { C.printf(c'LINUX PROCSELF STATE DETAIL: path=%s actual=%c expected=%c description=%d\n', path, i32(actual), i32(wanted), i32(description)); C.fflush(nil) }
		return valid
	}
}
fn state_matrix(bank &Handles, ordinary bool, reader i32, worker_tid i32, verbose bool) bool {
	unsafe {
		mut records := i32(0); mut failures := i32(0); pid := C.getpid()
		for view in 0 .. 3 { for form in 0 .. 5 { for format in 0 .. 2 {
			mut path := [96]char{}; suffix := if format == 0 { &char(c'stat') } else { &char(c'status') }
			if form == 0 { C.snprintf(&path[0], 96, c'self/%s', suffix) }
			else if form == 1 { C.snprintf(&path[0], 96, c'thread-self/%s', suffix) }
			else if form == 2 { C.snprintf(&path[0], 96, c'%d/%s', pid, suffix) }
			else { C.snprintf(&path[0], 96, c'%d/task/%d/%s', pid, if form == 3 { pid } else { worker_tid }, suffix) }
			wanted := if form == 1 { u8(`R`) } else if form == 4 { if reader == 0 { u8(`S`) } else { u8(`R`) } } else { if reader == 0 { u8(`R`) } else { u8(`S`) } }
			records++; if !state_record(bank.dirs[view], &path[0], format == 1, wanted, verbose) { failures++ }
		} } }
		if verbose { C.printf(c'LINUX PROCSELF STATES: ordinary=%d reader=%d records=%d failures=%d\n', i32(ordinary), reader, records, failures); C.fflush(nil) }
		return failures == 0
	}
}
fn state_worker(arg voidptr) voidptr {
	unsafe {
		mut state := &StateWorker(arg); C.__atomic_store_n(&state.tid, i32(C.syscall(isize(C.SYS_gettid))), 3)
		mut token := char(0)
		if C.write(state.report[1], &token, 1) != 1 || C.read(state.gate[0], &token, 1) != 1 { C._exit(1) }
		// Main blocks on the report pipe before these reads; this delay lets
		// the actual blocking read settle on either SMP or a single CPU.
		C.usleep(50000)
		state.failed = !state_matrix(state.bank, state.ordinary, 1, C.__atomic_load_n(&state.tid, 2), true)
		token = if state.failed { char(1) } else { char(0) }
		if C.write(state.report[1], &token, 1) != 1 || C.read(state.gate[0], &token, 1) != 1 { C._exit(1) }
		return nil
	}
}
fn stopped_zombie(bank &Handles, ordinary bool) bool {
	unsafe {
		child := C.fork(); if child < 0 { return false }
		if child == 0 { C.alarm(0); for { C.pause() } }
		mut reaped := false; mut outcome := i32(-1)
		defer { if !reaped { C.kill(child, C.SIGKILL); if C.waitpid(child, &outcome, 0) != child { C._exit(1) } } }
		if C.kill(child, C.SIGSTOP) != 0 || C.waitpid(child, &outcome, C.WUNTRACED) != child || C.WIFSTOPPED(outcome) == 0 || C.WSTOPSIG(outcome) != C.SIGSTOP { return false }
		mut good := true
		for code in [u8(`T`), `Z`]! {
			if code == `Z` {
				mut info := C.siginfo_t{}
				if C.kill(child, C.SIGCONT) != 0 || C.kill(child, C.SIGKILL) != 0 || C.waitid(C.P_PID, u32(child), &info, C.WEXITED | C.WNOWAIT) != 0 { return false }
			}
			mut failures := i32(0); mut records := i32(0)
			// Vinix removes zombie Thread objects before wait publishes; only
			// the process path remains. Zombie task-directory support is separate.
			for view in 0 .. 3 { for task in 0 .. if code == `Z` { 1 } else { 2 } { for format in 0 .. 2 {
				mut path := [96]char{}; suffix := if format == 0 { &char(c'stat') } else { &char(c'status') }
				if task == 0 { C.snprintf(&path[0], 96, c'%d/%s', child, suffix) } else { C.snprintf(&path[0], 96, c'%d/task/%d/%s', child, child, suffix) }
				records++; if !state_record(bank.dirs[view], &path[0], format == 1, code, true) { failures++ }
			} } }
			C.printf(c'LINUX PROCSELF CHILDSTATE: ordinary=%d state=%c records=%d failures=%d\n', i32(ordinary), i32(code), records, failures); C.fflush(nil)
			if failures != 0 { good = false }
		}
		if C.waitpid(child, &outcome, 0) != child { return false }; reaped = true
		return good && C.WIFSIGNALED(outcome) != 0 && C.WTERMSIG(outcome) == C.SIGKILL
	}
}
fn state_test(bank &Handles, ordinary bool) bool {
	unsafe {
		mut gate := [i32(-1), -1]!; mut report := [i32(-1), -1]!
		if C.pipe(&gate[0]) != 0 { return false }
		if C.pipe(&report[0]) != 0 { for fd in gate { C.close(fd) }; return false }
		mut state := StateWorker{bank: bank, ordinary: ordinary, gate: gate, report: report}; mut helper := C.pthread_t{}
		if C.pthread_create(&helper, nil, state_worker, &state) != 0 { for fd in gate { C.close(fd) }; for fd in report { C.close(fd) }; return false }
		mut token := char(0); if C.read(report[0], &token, 1) != 1 { C._exit(1) }; C.usleep(50000)
		tid := C.__atomic_load_n(&state.tid, 2); if tid <= 0 || tid == C.getpid() { C._exit(1) }
		mut good := state_matrix(bank, ordinary, 0, tid, true)
		if C.write(gate[1], &token, 1) != 1 || C.read(report[0], &token, 1) != 1 { C._exit(1) }
		if token != 0 { good = false }; C.usleep(50000)
		if good && !storage(bank, ordinary, tid) { good = false }
		if C.write(gate[1], &token, 1) != 1 || C.pthread_join(helper, nil) != 0 { C._exit(1) }
		for fd in gate { C.close(fd) }; for fd in report { C.close(fd) }
		return good && !state.failed
	}
}
fn cycle(bank &Handles, sleeper_tid i32) bool {
	unsafe {
		for view in 0 .. 3 { for kind in 0 .. 2 { for form in 0 .. 5 { for size in [i32(1), 64]! { for destination in 0 .. 2 {
			if !probe(bank, i32(view), i32(kind), i32(form), size, i32(destination), false) { return false }
		} } } }; for kind in 0 .. 2 { if !status(bank.dirs[view], i32(kind), false) { return false } } }
		return state_matrix(bank, false, 0, sleeper_tid, false)
	}
}
fn snapshot(fd i32, destination &Slabs) bool {
	unsafe {
		mut counts := &Slabs(destination); mut data := [4096]char{}
		length := C.pread(fd, &data[0], 4095, 0); if length <= 0 || length >= 4096 { return false }; data[int(length)] = 0
		mut position := 0; mut large := false
		for position < int(length) {
			mut size := u64(0); mut duplicate := u64(0); mut live := u64(0); mut pages := u64(0)
			if C.sscanf(&data[position], c'size-%lu %lu %lu %lu', &size, &duplicate, &live, &pages) == 4 {
				if counts.rows >= 32 || size != duplicate { return false }; counts.sizes[counts.rows] = size; counts.live[counts.rows] = live; counts.rows++
			} else if C.sscanf(&data[position], c'large - - %lu', &counts.large) == 1 { large = true }
			for position < int(length) && data[position] != char(10) { position++ }; position++
		}
		return counts.rows > 0 && large
	}
}
fn storage(bank &Handles, ordinary bool, sleeper_tid i32) bool {
	unsafe {
		observer := C.open(c'/proc/slabinfo', C.O_RDONLY); if observer < 0 { return false }; defer { C.close(observer) }
		for _ in 0 .. 32 { if !cycle(bank, sleeper_tid) { return false } }; C.usleep(13000000)
		mut before := &Slabs(C.alloca(usize(sizeof(Slabs)))); *before = Slabs{}; if !snapshot(observer, before) { return false }
		for _ in 0 .. 200 { if !cycle(bank, sleeper_tid) { return false } }; C.usleep(13000000)
		mut after := &Slabs(C.alloca(usize(sizeof(Slabs)))); *after = Slabs{}; if !snapshot(observer, after) || before.rows != after.rows { return false }
		mut valid := before.large == after.large
		for index in 0 .. int(before.rows) {
			if before.sizes[index] != after.sizes[index] || before.live[index] != after.live[index] { valid = false }
			C.printf(c'LINUX PROCSELF SLAB: ordinary=%d size=%lu live=%lu/%lu cycles=200\n', i32(ordinary), before.sizes[index], before.live[index], after.live[index])
		}
		C.printf(c'LINUX PROCSELF STORAGE: ordinary=%d cycles=200 big_pages=%lu/%lu %s\n', i32(ordinary), before.large, after.large, if valid { &char(c'PASS') } else { &char(c'FAIL') }); C.fflush(nil)
		return valid
	}
}
fn chroot_test() bool {
	unsafe {
		child := C.fork(); if child < 0 { return false }
		if child == 0 {
			if C.chroot(c'/tmp/linux-procself/jail') != 0 || C.chdir(c'/') != 0 { C._exit(1) }
			fd := C.open(c'/view', C.O_PATH | C.O_DIRECTORY); if fd < 0 { C._exit(1) }
			missing := C.open(c'/proc', C.O_PATH); if missing >= 0 { C.close(missing); C.close(fd); C._exit(1) }
			valid := status(fd, 0, true) && status(fd, 1, true); C.close(fd)
			C.printf(c'LINUX PROCSELF CHROOT: no-/proc relative-self/thread-self-status %s\n', if valid { &char(c'PASS') } else { &char(c'FAIL') }); C.fflush(nil)
			C._exit(if valid { 0 } else { 1 })
		}
		mut outcome := i32(-1); return C.waitpid(child, &outcome, 0) == child && outcome == 0
	}
}
fn namespace_identity(fd i32, local bool, outer_pid i32) bool {
	unsafe {
		mut link := [64]char{}; mut outer_tid := i32(-1); mut parsed_pid := i32(-1)
		length := C.readlinkat(fd, c'thread-self', &link[0], 63)
		if length <= 0 || length >= 64 { return false }; link[int(length)] = 0
		if C.sscanf(&link[0], c'%d/task/%d', &parsed_pid, &outer_tid) != 2 { return false }
		local_tid := i32(C.syscall(isize(C.SYS_gettid)))
		if parsed_pid != if local { 1 } else { outer_pid } || (local && outer_tid != local_tid) { return false }
		status_fd := C.openat(fd, c'thread-self/status', C.O_RDONLY); if status_fd < 0 { return false }; defer { C.close(status_fd) }
		mut data := [4096]char{}; n := C.read(status_fd, &data[0], 4095); if n <= 0 || n >= 4096 { return false }; data[int(n)] = 0
		pid_line := C.strstr(&data[0], c'\nPid:'); tgid_line := C.strstr(&data[0], c'\nTgid:'); ns_line := C.strstr(&data[0], c'\nNSpid:')
		if pid_line == nil || tgid_line == nil || ns_line == nil { return false }
		mut pid := i32(-1); mut tgid := i32(-1); mut first := i32(-1); mut second := i32(-1); mut third := i32(-1)
		if C.sscanf(pid_line, c'\nPid: %d', &pid) != 1 || C.sscanf(tgid_line, c'\nTgid: %d', &tgid) != 1 { return false }
		count := C.sscanf(ns_line, c'\nNSpid: %d %d %d', &first, &second, &third)
		policy := i32(C.syscall(isize(C.SYS_sched_getscheduler), isize(0))); mut param := C.sched_param{}
		if policy < 0 || C.syscall(isize(C.SYS_sched_getparam), isize(0), &param) != 0 { return false }
		mut numeric := [96]char{}; C.snprintf(&numeric[0], 96, c'%s/stat', &link[0])
		alias_valid := stat_record(fd, c'thread-self/stat', outer_tid, policy, param.sched_priority, true, true)
		numeric_valid := stat_record(fd, &numeric[0], outer_tid, policy, param.sched_priority, true, true)
		valid := pid == outer_tid && tgid == parsed_pid && first == outer_tid && count == if local { 1 } else { 2 } && (local || second == local_tid) && alias_valid && numeric_valid
		C.printf(c'LINUX PROCSELF TASKIDS: local=%d leader=%d pid=%d/%d tgid=%d/%d nspid=%d:%d count=%d expected_count=%d %s\n', i32(local), i32(local_tid == 1), pid, outer_tid, tgid, parsed_pid, first, second, count, if local { 1 } else { 2 }, if valid { &char(c'PASS') } else { &char(c'FAIL') }); C.fflush(nil)
		return valid
	}
}
fn namespace_worker(arg voidptr) voidptr {
	unsafe {
		mut state := &NamespaceWorker(arg)
		if C.getpid() != 1 || C.syscall(isize(C.SYS_gettid)) <= 1 { state.failed = true; return nil }
		if !scheduling(i32(C.SCHED_FIFO), 23) { state.failed = true; return nil }
		outer := namespace_identity(state.outer, false, state.outer_pid); inner := namespace_identity(state.inner, true, state.outer_pid)
		state.failed = !outer || !inner; if !scheduling(i32(C.SCHED_OTHER), 0) { state.failed = true }; return nil
	}
}
fn namespace_child(outer i32) i32 {
	unsafe {
		// Keep the inherited proc mount uncovered so its directory descriptor
		// continues to select the root view while testing the new tree.
		if C.getpid() != 1 || C.mkdir(c'/tmp/linux-procself/ns-view', u32(0o755)) != 0
			|| C.mount(c'proc', c'/tmp/linux-procself/ns-view', c'proc', 0, nil) != 0 { return 1 }
		inner := C.open(c'/tmp/linux-procself/ns-view', C.O_PATH | C.O_DIRECTORY); if inner < 0 { return 1 }; defer { C.close(inner) }
		mut link := [64]char{}; n := C.readlinkat(outer, c'self', &link[0], 63); if n <= 0 || n >= 64 { return 1 }; link[int(n)] = 0
		mut outer_pid := i32(-1); if C.sscanf(&link[0], c'%d', &outer_pid) != 1 || outer_pid <= 1 { return 1 }
		mut valid := namespace_identity(outer, false, outer_pid) && namespace_identity(inner, true, outer_pid)
		mut state := NamespaceWorker{outer: outer, inner: inner, outer_pid: outer_pid}; mut helper := C.pthread_t{}
		if C.pthread_create(&helper, nil, namespace_worker, &state) != 0 { return 1 }
		if C.pthread_join(helper, nil) != 0 { C._exit(1) }; if state.failed { valid = false }
		C.printf(c'LINUX PROCSELF PIDVIEW: outer-and-own leader/worker Pid/Tgid/NSpid %s\n', if valid { &char(c'PASS') } else { &char(c'FAIL') }); C.fflush(nil)
		return if valid { 0 } else { 1 }
	}
}
fn namespace_test(outer i32) bool {
	unsafe {
		parent := C.fork(); if parent < 0 { return false }
		if parent == 0 {
			if C.unshare(C.CLONE_NEWNS) != 0 || C.unshare(C.CLONE_NEWPID) != 0 { C._exit(1) }
			child := C.fork(); if child < 0 { C._exit(1) }; if child == 0 { C._exit(namespace_child(outer)) }
			mut outcome := i32(-1); C._exit(if C.waitpid(child, &outcome, 0) == child && outcome == 0 { 0 } else { 1 })
		}
		mut outcome := i32(-1); return C.waitpid(parent, &outcome, 0) == parent && outcome == 0
	}
}
fn qualify(bank &Handles, ordinary bool) bool {
	unsafe {
		defer { cleanup(bank) }
		if ordinary && (C.setgroups(0, nil) != 0 || C.setgid(1000) != 0 || C.setuid(1000) != 0) { return false }
		C.alarm(300); if !workload(0) { return false }; mut good := matrix(bank, ordinary, 0)
		mut states := [Worker{bank: bank, ordinary: ordinary, slot: 1}, Worker{bank: bank, ordinary: ordinary, slot: 2}]!
		mut threads := [2]C.pthread_t{}; mut started := 0
		for index in 0 .. 2 { if C.pthread_create(&threads[index], nil, worker, &states[index]) != 0 { good = false; break }; started++ }
		for index in 0 .. started {
			// A failed join cannot unwind storage that a worker may still use.
			if C.pthread_join(threads[index], nil) != 0 { C._exit(1) }
			if states[index].failed || states[index].tid == C.getpid() { good = false }
		}
		if started != 2 || states[0].tid == states[1].tid { good = false }
		if !ordinary && !chroot_test() { good = false }
		if !ordinary && !namespace_test(bank.dirs[0]) { good = false }
		if !stopped_zombie(bank, ordinary) { good = false }
		if !good || !state_test(bank, ordinary) { return false }
		C.alarm(0); return true
	}
}
@[export: 'main']
pub fn run() i32 {
	unsafe {
		C.puts(c'LINUX PROCSELF SETUP: rtprio limit'); C.fflush(nil)
		param := C.sched_param{}; mut scratch := C.sched_param{}; e := C.__errno_location()
		*e = 0; if C.sched_setscheduler(0, C.SCHED_OTHER, &param) != -1 || *e != C.ENOSYS { C.puts(c'LINUX PROCSELF FAIL: musl sched_setscheduler refusal'); C.fflush(nil); return 1 }
		*e = 0; if C.sched_getscheduler(0) != -1 || *e != C.ENOSYS { C.puts(c'LINUX PROCSELF FAIL: musl sched_getscheduler refusal'); C.fflush(nil); return 1 }
		*e = 0; if C.sched_getparam(0, &scratch) != -1 || *e != C.ENOSYS { C.puts(c'LINUX PROCSELF FAIL: musl sched_getparam refusal'); C.fflush(nil); return 1 }
		C.puts(c'LINUX PROCSELF MUSL SCHED: process wrappers refuse ENOSYS'); C.fflush(nil)
		*e = 0
		limit := C.rlimit{rlim_cur: 99, rlim_max: 99}; result := C.setrlimit(C.RLIMIT_RTPRIO, &limit)
		C.printf(c'LINUX PROCSELF SETUP: setrlimit=%d errno=%d\n', result, *C.__errno_location()); C.fflush(nil)
		if result != 0 { C.puts(c'LINUX PROCSELF FAIL: rtprio limit'); C.fflush(nil); return 1 }
		C.puts(c'LINUX PROCSELF SETUP: ordinary policy'); C.fflush(nil)
		if !scheduling(i32(C.SCHED_OTHER), 0) { C.printf(c'LINUX PROCSELF FAIL: ordinary policy errno=%d\n', *C.__errno_location()); C.fflush(nil); return 1 }
		if C.mkdir(c'/tmp/linux-procself', u32(0o755)) != 0 || C.mkdir(c'/tmp/linux-procself/alias', u32(0o755)) != 0
			|| C.mkdir(c'/tmp/linux-procself/jail', u32(0o755)) != 0 || C.mkdir(c'/tmp/linux-procself/jail/view', u32(0o755)) != 0
			|| C.mount(c'proc', c'/tmp/linux-procself/alias', c'proc', 0, nil) != 0 || C.mount(c'proc', c'/tmp/linux-procself/jail/view', c'proc', 0, nil) != 0 { return 1 }
		mut bank := &Handles(C.alloca(usize(sizeof(Handles)))); *bank = Handles{}
		for index in 0 .. 3 { bank.dirs[index] = -1 }; for index in 0 .. 6 { bank.links[index] = -1 }; defer { cleanup(bank) }
		if !prepare(bank) { return 1 }; mut failed := false
		for ordinary in [true, false]! {
			C.fflush(nil); child := C.fork(); if child < 0 { return 1 }; if child == 0 { C._exit(if qualify(bank, ordinary) { 0 } else { 1 }) }
			mut outcome := i32(-1); if C.waitpid(child, &outcome, 0) != child || outcome != 0 { failed = true }
		}
		if failed { C.puts(c'LINUX PROCSELF FAIL: relative target, reader identity, resolution or storage'); C.fflush(nil); return 1 }
		C.puts(c'LINUX PROCSELF: PASS'); C.fflush(nil); for { C.pause() }
	}
	return 0
}
